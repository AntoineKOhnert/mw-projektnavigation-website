# MW Projekt Navigation – saubere URLs ohne .html
# Auf dem Server in ".htaccess" umbenennen (mit Punkt vorne, ohne .txt)

Options -MultiViews
DirectoryIndex index.html
RewriteEngine On

# /index.html -> /
RewriteCond %{THE_REQUEST} \s/+index\.html[\s?] [NC]
RewriteRule ^index\.html$ / [R=301,L]

# /seite.html -> /seite (alte Links leiten dauerhaft weiter)
RewriteCond %{THE_REQUEST} \s/+([^\s?]+)\.html[\s?] [NC]
RewriteRule ^ /%1 [R=301,L,NE]

# /seite/ -> /seite (Schrägstrich am Ende entfernen)
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^(.+)/$ /$1 [R=301,L]

# /seite -> intern seite.html ausliefern
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^(.+)$ $1.html [L]
